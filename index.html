from fastapi import FastAPI, HTTPException from pydantic import BaseModel from typing import List, Dict from ortools.sat.python import cp_model app = FastAPI(title="Enterprise Nurse Scheduling API", version="1.0") # --- Data Models --- class Nurse(BaseModel): id: int name: str grade: str # เช่น Senior, Junior (ใช้ควบคุมสัดส่วนในเวรได้) class SchedulingConfig(BaseModel): num_days: int = 7 # จัดตารางล่วงหน้ากี่วัน (เช่น 7 วัน, 30 วัน) nurses: List[Nurse] # กำหนดความต้องการขั้นต่ำของพยาบาลในแต่ละกะ [เช้า, บ่าย, ดึก] shift_requirements: Dict[str, int] = {"M": 2, "A": 2, "N": 1} # --- Solver Core Logic --- def generate_schedule_logic(config: SchedulingConfig): num_nurses = len(config.nurses) num_days = config.num_days shifts = ["M", "A", "N", "O"] # M: Morning, A: Afternoon, N: Night, O: Off num_shifts = len(shifts) model = cp_model.CpModel() # 1. สร้างตัวแปรการตัดสินใจ (Decision Variables) # work[n, d, s] จะเป็น 1 ถ้าพยาบาล n ทำงานในวัน d กะ s, นอกนั้นเป็น 0 work = {} for n in range(num_nurses): for d in range(num_days): for s in range(num_shifts): work[n, d, s] = model.NewBoolVar(f'work_{n}_{d}_{shifts[s]}') # 2. Hard Constraints (เงื่อนไขที่ "ต้อง" ปฏิบัติตามอย่างเคร่งครัด) # เงื่อนไขข้อที่ 1: พยาบาล 1 คน เข้าได้แค่ 1 กะต่อวันเท่านั้น (รวมวันหยุดด้วย) for n in range(num_nurses): for d in range(num_days): model.Add(sum(work[n, d, s] for s in range(num_shifts)) == 1) # เงื่อนไขข้อที่ 2: ควบคุมความปลอดภัยของคนไข้และพยาบาล (Shift Transition Constraints) # ห้ามเข้าเวร ดึก (N) แล้วต่อด้วย เช้า (M) หรือ บ่าย (A) ในวันถัดไปทันที เพื่อป้องกันการล้า for n in range(num_nurses): for d in range(num_days - 1): # เข้าเวรดึก (index 2: N) วันนี้ วันพรุ่งนี้ห้ามเข้าเช้า (index 0: M) หรือบ่าย (index 1: A) model.Add(work[n, d, 2] + work[n, d + 1, 0] <= 1) model.Add(work[n, d, 2] + work[n, d + 1, 1] <= 1) # เงื่อนไขข้อที่ 3: จำนวนพยาบาลในแต่ละกะต้องครบตามที่แผนกต้องการ for d in range(num_days): # กะเช้า (M) model.Add(sum(work[n, d, 0] for n in range(num_nurses)) >= config.shift_requirements["M"]) # กะบ่าย (A) model.Add(sum(work[n, d, 1] for n in range(num_nurses)) >= config.shift_requirements["A"]) # กะดึก (N) model.Add(sum(work[n, d, 2] for n in range(num_nurses)) >= config.shift_requirements["N"]) # 3. Soft Constraints / Objectives (เงื่อนไขที่พยายามทำให้ดีที่สุด เช่น ความเป็นธรรม) # ตัวอย่าง: กระจายเวรดึกให้ทุกคนเท่าๆ กันมากที่สุด # ในระดับ Enterprise จะใช้การตั้ง Objective Function เพื่อ Minimize ความเหลื่อมล้ำ min_shifts_per_nurse = (num_days * config.shift_requirements["N"]) // num_nurses max_shifts_per_nurse = min_shifts_per_nurse + 1 for n in range(num_nurses): num_night_shifts = sum(work[n, d, 2] for d in range(num_days)) model.Add(num_night_shifts >= min_shifts_per_nurse) model.Add(num_night_shifts <= max_shifts_per_nurse) # 4. สั่งให้ Solver ทำงาน solver = cp_model.CpSolver() solver.parameters.max_time_in_seconds = 30.0 # จำกัดเวลาคำนวณไว้ที่ 30 วินาที (สเกลใหญ่ต้องคุมเวลา) status = solver.Solve(model) # 5. รวบรวมผลลัพธ์ if status == cp_model.OPTIMAL or status == cp_model.FEASIBLE: schedule_result = [] for d in range(num_days): day_schedule = {"day": d + 1, "assignments": []} for n in range(num_nurses): for s in range(num_shifts): if solver.BooleanValue(work[n, d, s]): if shifts[s] != "O": # ไม่แสดงวันหยุดในตารางทำงานหลัก หรือแสดงเป็น OFF ก็ได้ day_schedule["assignments"].append({ "nurse_id": config.nurses[n].id, "nurse_name": config.nurses[n].name, "shift": shifts[s] }) schedule_result.append(day_schedule) return {"status": "Success", "schedule": schedule_result} else: return {"status": "Infeasible", "message": "ไม่สามารถจัดตารางตามเงื่อนไขที่กำหนดได้เนื่องจากทรัพยากรไม่พอ"} # --- API Endpoint --- @app.post("/api/v1/schedule/generate") def generate_schedule(config: SchedulingConfig): try: result = generate_schedule_logic(config) if result["status"] == "Infeasible": raise HTTPException(status_code=400, detail=result["message"]) return result except Exception as e: raise HTTPException(status_code=500, detail=str(e)) if __name__ == "__main__": import uvicorn uvicorn.run(app, host="0.0.0.0", port=8000)
